cmake_minimum_required(VERSION 3.24)
project(coremio_database_unit_test C)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(GLOB COREMIO_LIBRARY_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/../src/*.c ${CMAKE_CURRENT_SOURCE_DIR}/../src/neural_networks/*.c)
file(GLOB TEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.c)
add_executable(coremio_database_unit_test ${COREMIO_LIBRARY_SOURCES} ${TEST_SOURCES})
target_include_directories(coremio_database_unit_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../include/coremio)
target_link_libraries(coremio_database_unit_test PRIVATE Threads::Threads SQLite3::SQLite3 m)
